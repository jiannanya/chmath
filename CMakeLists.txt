cmake_minimum_required(VERSION 3.20)
project(chmath VERSION 2.0.0 DESCRIPTION "Compact C++20 mathematics for graphics, games and CAD" LANGUAGES CXX)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CHMATH_IS_TOP_LEVEL OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(CHMATH_IS_TOP_LEVEL ON)
endif()
option(CHMATH_BUILD_TESTS "Build tests and standalone header checks" ${CHMATH_IS_TOP_LEVEL})
option(CHMATH_BUILD_EXAMPLES "Build examples" ${CHMATH_IS_TOP_LEVEL})
option(CHMATH_BUILD_BENCHMARKS "Build reproducible microbenchmarks" OFF)
option(CHMATH_ENABLE_SIMD "Enable compile-time SSE2/NEON batch and mat4 kernels where supported" ON)
option(CHMATH_SANITIZERS "Enable ASan and UBSan for project validation targets (GCC/Clang)" OFF)
option(CHMATH_WARNINGS_AS_ERRORS "Treat warnings in project validation targets as errors" ON)
option(CHMATH_COVERAGE "Enable Clang source-based coverage on project validation targets" OFF)
option(CHMATH_PERFORMANCE_CHECKS "Enforce relative performance thresholds on a controlled runner" OFF)

add_library(chmath INTERFACE)
add_library(chm::chmath ALIAS chmath)
target_compile_features(chmath INTERFACE cxx_std_20)
target_include_directories(chmath INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(chmath INTERFACE CHMATH_ENABLE_SIMD=$<BOOL:${CHMATH_ENABLE_SIMD}>)

function(chmath_validate target)
    target_link_libraries(${target} PRIVATE chm::chmath)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_definitions(${target} PRIVATE CHMATH_PERFORMANCE_CHECKS=$<BOOL:${CHMATH_PERFORMANCE_CHECKS}>)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" OR CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
        target_compile_options(${target} PRIVATE /W4 /permissive- /Zc:__cplusplus /utf-8)
        if(CHMATH_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(CHMATH_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
    if(CHMATH_SANITIZERS)
        if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR WIN32)
            message(FATAL_ERROR "CHMATH_SANITIZERS requires GCC/Clang on a Unix-like platform")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
    if(CHMATH_COVERAGE)
        if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
            message(FATAL_ERROR "CHMATH_COVERAGE requires Clang")
        endif()
        target_compile_options(${target} PRIVATE -fprofile-instr-generate -fcoverage-mapping)
        target_link_options(${target} PRIVATE -fprofile-instr-generate)
    endif()
endfunction()

if(CHMATH_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    file(GLOB test_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_*.cpp")
    set(chmath_case_count 0)
    set(chmath_case_names "")
    foreach(source IN LISTS test_sources)
        get_filename_component(name "${source}" NAME_WE)
        add_executable(${name} "${source}")
        chmath_validate(${name})
        if(name STREQUAL "test_extended_stress")
            target_link_libraries(${name} PRIVATE Threads::Threads)
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
        file(STRINGS "${source}" declared_cases REGEX "^[ \t]*CH_TEST\\([A-Za-z0-9_]+\\)")
        if(declared_cases)
            foreach(declaration IN LISTS declared_cases)
                string(REGEX REPLACE ".*CH_TEST\\(([A-Za-z0-9_]+)\\).*" "\\1" case_name "${declaration}")
                if(case_name IN_LIST chmath_case_names)
                    message(FATAL_ERROR "Duplicate test case: ${case_name}")
                endif()
                list(APPEND chmath_case_names "${case_name}")
                add_test(NAME "${name}.${case_name}" COMMAND ${name} --case "${case_name}")
                set_tests_properties("${name}.${case_name}" PROPERTIES LABELS "${name}" TIMEOUT 60)
                if(case_name MATCHES "^perf_")
                    set_tests_properties("${name}.${case_name}" PROPERTIES LABELS "performance" RUN_SERIAL TRUE)
                elseif(case_name MATCHES "^stress_")
                    set_tests_properties("${name}.${case_name}" PROPERTIES LABELS "stress" TIMEOUT 120)
                elseif(case_name MATCHES "^safety_")
                    set_tests_properties("${name}.${case_name}" PROPERTIES LABELS "safety" TIMEOUT 60)
                endif()
                math(EXPR chmath_case_count "${chmath_case_count}+1")
            endforeach()
        else()
            add_test(NAME ${name} COMMAND ${name})
        endif()
    endforeach()
    message(STATUS "chmath: ${chmath_case_count} independently runnable test cases")
    if(chmath_case_count LESS 1000)
        message(FATAL_ERROR "The full test suite must contain at least 1000 named cases")
    endif()
    if(CHMATH_COVERAGE)
        # One translation unit avoids LLVM hash=0 mappings for unused inline definitions
        # when combining coverage from independent header-only test executables.
        set(coverage_source "")
        foreach(source IN LISTS test_sources)
            get_filename_component(name "${source}" NAME_WE)
            if(NOT name STREQUAL "test_allocation")
                string(APPEND coverage_source "#include \"${source}\"\n")
            endif()
        endforeach()
        file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/coverage_all.cpp" CONTENT "${coverage_source}")
        add_executable(chmath_coverage "${CMAKE_CURRENT_BINARY_DIR}/coverage_all.cpp")
        chmath_validate(chmath_coverage)
        target_link_libraries(chmath_coverage PRIVATE Threads::Threads)
        add_test(NAME chmath_coverage COMMAND chmath_coverage)
        set_tests_properties(chmath_coverage PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
    endif()
    file(GLOB_RECURSE public_headers CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/chmath/*.hpp")
    set(header_sources "")
    foreach(header IN LISTS public_headers)
        file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}/src" "${header}")
        string(MAKE_C_IDENTIFIER "${relative}" identifier)
        set(source "${CMAKE_CURRENT_BINARY_DIR}/header_checks/${identifier}.cpp")
        file(GENERATE OUTPUT "${source}" CONTENT "#include <${relative}>\n")
        list(APPEND header_sources "${source}")
    endforeach()
    add_library(chmath_header_checks OBJECT ${header_sources})
    chmath_validate(chmath_header_checks)
    add_executable(chmath_odr tests/odr/main.cpp tests/odr/other.cpp)
    chmath_validate(chmath_odr)
    add_test(NAME chmath_odr COMMAND chmath_odr)
    add_test(NAME chmath_install_consumer
        COMMAND ${CMAKE_COMMAND}
            -DCHMATH_SOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}
            -DCHMATH_BINARY_DIR=${CMAKE_CURRENT_BINARY_DIR}
            -DCHMATH_CXX_COMPILER=${CMAKE_CXX_COMPILER}
            -DCHMATH_GENERATOR=${CMAKE_GENERATOR}
            -DCHMATH_CONFIG=$<CONFIG>
            -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/test_install.cmake)
endif()
